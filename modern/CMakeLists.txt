cmake_minimum_required(VERSION 3.25)
project(D2Next LANGUAGES C CXX)
# CMake 3.28+ scans C++20 TUs for modules. We don't use modules, and Linux clang packages often omit clang-scan-deps.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  if(EMSCRIPTEN)
    message(FATAL_ERROR "The web build is wasm64. Configure with -sMEMORY64=1 (see modern/web/README.md).")
  else()
    message(FATAL_ERROR "D2Next is 64-bit only. On Windows configure with -A x64.")
  endif()
endif()
# Other systems (Android, Linux, the web) have one way to run: the game linked in, SDL for the window, OpenGL ES to draw.
if(NOT WIN32)
  set(D2MODERN_INPROCESS ON CACHE BOOL "" FORCE)
  set(D2MODERN_PLATFORM "SDL" CACHE STRING "" FORCE)
  set(D2MODERN_RENDERER "GL" CACHE STRING "" FORCE)
  # The part of Windows.h D2MOO's headers use (../external/win32compat), and the compiler settings its code needs (see
  # ../CMakeLists.txt), for the native code here that includes those headers.
  include_directories(BEFORE SYSTEM ${CMAKE_CURRENT_SOURCE_DIR}/../external/win32compat/include)
  add_compile_options(-fsigned-char -fms-extensions -Wno-microsoft-cast -Wno-microsoft-anon-tag -Wno-ignored-attributes
    -Wno-ignored-pragmas -Wno-unknown-pragmas -Wno-pragma-pack -Wno-multichar)
  if(EMSCRIPTEN)
    # Native wasm exceptions: JS invoke_* wrappers mis-type function pointers on wasm64.
    # MEMORY64 must match the flags emcmake was given so every translation unit is wasm64.
    add_compile_options(-sMEMORY64=1 -fwasm-exceptions)
    add_link_options(-sMEMORY64=1 -fwasm-exceptions)
  endif()
endif()

set(STORM_SKIP_INSTALL ON CACHE BOOL "" FORCE)

  set(STORM_UNICODE ON CACHE BOOL "" FORCE)

  set(STORM_UNICODE OFF CACHE BOOL "" FORCE)
endif()
# Where the game (D2MOO's D2Game and D2Common) runs. By default D2MOO is built here as static libraries and linked
# into D2Next.exe (run it with --native): no DLLs and no second process. With D2MODERN_INPROCESS=OFF it runs in
# NativeGameHost.exe, a separate process that loads the x64 DLLs built by Build-X64.cmd (--native-town DLL_DIR); the
# x64 DLLs are what gets compared with the original 1.10f ones, so that build stays for checking (out/modern-dev).
option(D2MODERN_INPROCESS "Link D2MOO's game libraries into D2Next instead of running NativeGameHost.exe" ON)
if(D2MODERN_INPROCESS)
  set(D2MOO_MODERN_WINDOWS_COMPAT ON CACHE BOOL "" FORCE)
  set(ENABLE_D2DETOURS_EMBEDDED_PATCHES OFF CACHE BOOL "" FORCE)
  set(D2MOO_WITH_STATIC_TESTS OFF CACHE BOOL "" FORCE)
  set(D2MOO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(D2MOO_INSTALL OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  # D2MOO also adds external/StormLib (target storm), with fourteen functions renamed so they do not clash with Storm's.
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/.. d2moo EXCLUDE_FROM_ALL)
  if(MSVC)
    set(D2MODERN_STORMLIB_RENAME "/FI${CMAKE_CURRENT_SOURCE_DIR}/../source/Storm/src/x64/StormLibRename.h")
  else()
    set(D2MODERN_STORMLIB_RENAME -include "${CMAKE_CURRENT_SOURCE_DIR}/../source/Storm/src/x64/StormLibRename.h")
  endif()
  # The network: the stand-in transport (a queue each way) instead of D2Net's sockets.
  set_property(TARGET D2Net PROPERTY SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/../tools/x64/MockD2Net.cpp)
  target_compile_definitions(D2Net PRIVATE MOCKD2NET_STATIC)
else()
  add_subdirectory(external/StormLib EXCLUDE_FROM_ALL)
  set(D2MODERN_STORMLIB_RENAME "")
endif()

  if(MSVC)

  target_link_libraries(d2native PRIVATE D2Game D2Common D2Net Fog Storm D2Lang D2CMP D2Hell)

if(WIN32)
  add_library(ModernHostNet SHARED ../tools/x64/MockD2Net.cpp)

  add_executable(NativeGameHost src/NativeGameHost.cpp src/NativeSession.cpp ../external/squall/storm/thread/CCritSect.cpp)
  target_compile_features(NativeGameHost PRIVATE cxx_std_20)
  # Legacy squall error declarations accept mutable format literals.
  target_compile_options(NativeGameHost PRIVATE /Zc:strictStrings-)
  target_include_directories(NativeGameHost PRIVATE ../external/squall)
  foreach(module D2Game D2Common D2CommonDefinitions Fog Storm D2Net D2Lang D2CMP D2Hell)
    target_include_directories(NativeGameHost PRIVATE ../source/${module}/include)
  endforeach()
  target_compile_definitions(NativeGameHost PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN WHOA_SYSTEM_WIN
    D2COMMON_DLL_DECL= D2GAME_DLL_DECL= FOG_DLL_DECL= STORM_DLL_DECL= D2NET_DLL_DECL= D2LANG_DLL_DECL= D2CMP_DLL_DECL= D2HELL_DLL_DECL=)
  target_link_libraries(NativeGameHost PRIVATE winmm)
  add_dependencies(NativeGameHost ModernHostNet)

# The window and its input: Win32 directly, or SDL3 (external/SDL), which also runs on Linux, Android and the web.
# Configure with -DD2MODERN_PLATFORM=SDL to use it on Windows. SDL is linked in, except on Android, where SDL's Java
# activity loads it as its own libSDL3.so.
set(D2MODERN_PLATFORM "Win32" CACHE STRING "Window and input layer: Win32 or SDL")
set_property(CACHE D2MODERN_PLATFORM PROPERTY STRINGS Win32 SDL)
if(D2MODERN_PLATFORM STREQUAL "SDL")
  if(ANDROID)
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  else()
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
  endif()
  if(EMSCRIPTEN)
    # Single-thread wasm: music loads on the game thread (Audio.cpp). A pthread build needs COOP/COEP headers.
    set(SDL_PTHREADS OFF CACHE BOOL "" FORCE)
  endif()
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory(external/SDL EXCLUDE_FROM_ALL)
  target_sources(D2Next PRIVATE src/PlatformSdl.cpp)
  if(ANDROID)
    target_link_libraries(D2Next PRIVATE SDL3::SDL3-shared)
  else()
    target_link_libraries(D2Next PRIVATE SDL3::SDL3-static)
  endif()
  target_compile_definitions(D2Next PRIVATE D2MODERN_SDL=1)
endif()
# What draws: Direct3D 11 (Windows), or OpenGL ES 3 (GL: the backend for Android and, as WebGL 2, the web; on desktops it
# uses OpenGL ES or OpenGL 3.3). GL needs the SDL platform.
set(D2MODERN_RENDERER "D3D11" CACHE STRING "Graphics backend: D3D11 or GL")
set_property(CACHE D2MODERN_RENDERER PROPERTY STRINGS D3D11 GL)
if(D2MODERN_RENDERER STREQUAL "GL")
  if(NOT D2MODERN_PLATFORM STREQUAL "SDL")
    message(FATAL_ERROR "D2MODERN_RENDERER=GL needs D2MODERN_PLATFORM=SDL")
  endif()
  target_sources(D2Next PRIVATE src/GlBackend.cpp)
  target_compile_definitions(D2Next PRIVATE D2MODERN_GL=1)
else()
  target_sources(D2Next PRIVATE src/D3D11Backend.cpp)
  target_link_libraries(D2Next PRIVATE d3d11 dxgi d3dcompiler)
endif()
if(EMSCRIPTEN)
  # wasm64 + WebGL 2. The HTML page mounts MPQs into the virtual filesystem, then callMain().
  # Native wasm exceptions; the browser main loop (emscripten_set_main_loop) yields. Asyncify
  # cannot run with memory64 + wasm exceptions (binaryen abort).
  set(D2MODERN_WEB_SHELL "${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html")
  target_link_options(D2Next PRIVATE
    "SHELL:-sMEMORY64=1"
    "SHELL:-sUSE_WEBGL2=1"
    "SHELL:-sMIN_WEBGL_VERSION=2"
    "SHELL:-sMAX_WEBGL_VERSION=2"
    "SHELL:-sFULL_ES3=1"
    "SHELL:-sALLOW_MEMORY_GROWTH=1"
    "SHELL:-sINITIAL_MEMORY=512MB"
    "SHELL:-sSTACK_SIZE=1MB"
    "SHELL:-sFORCE_FILESYSTEM=1"
    "SHELL:-sINVOKE_RUN=0"
    "SHELL:-sENVIRONMENT=web"
    "SHELL:-sEXPORTED_RUNTIME_METHODS=callMain,FS,PATH,UTF8ToString"
    "SHELL:-sEXPORTED_FUNCTIONS=_main"
    "SHELL:-sNO_EXIT_RUNTIME=1"
    "SHELL:--profiling-funcs"
    "SHELL:-lidbfs.js"
    -fwasm-exceptions
    --shell-file "${D2MODERN_WEB_SHELL}")
  set_target_properties(D2Next PROPERTIES SUFFIX ".html")
  set_property(TARGET D2Next APPEND PROPERTY LINK_DEPENDS "${D2MODERN_WEB_SHELL}")
endif()
